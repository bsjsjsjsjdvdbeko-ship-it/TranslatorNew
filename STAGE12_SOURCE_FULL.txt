class SettingsRepository(context: Context) {
    private val prefs = try {

        EncryptedSharedPreferences.create(context,"sari_secure_settings",master,EncryptedSharedPreferences.PrefKeyEncryptionScheme.AES256_SIV,EncryptedSharedPreferences.PrefValueEncryptionScheme.AES256_GCM)
    } catch (_:Exception){null}

    suspend fun save(provider:Int,speechKey:String,translationKey:String,source:String,target:String,font:Int,opacity:Int,email:String,speechLanguage:String,contextAwareTranslation:Boolean=true,translationMemoryEnabled:Boolean=true,glossary:String="",translationQuality:Int=QUALITY_HUMAN,sttProvider:Int=STT_GROQ,deepgramKey:String="",openAiKey:String="",contextPrompt:String="",subtitlePosition:Int=POS_BOTTOM,calibrationMs:Long=0L)=withContext(Dispatchers.IO){
        val p=prefs ?: throw IllegalStateException("تعذر تفعيل التخزين المشفر على هذا الجهاز.")
        p.edit().putInt("provider",provider.coerceIn(PROVIDER_GROQ,PROVIDER_OPENAI)).putBoolean("providerSchemaV3",true)
            .putString("speechKey",speechKey).putString("translationKey",translationKey).putString("source",source).putString("target",target)
            .putInt("font",font.coerceIn(12,48)).putInt("opacity",opacity.coerceIn(0,100)).putString("email",email).putString("speechLanguage",speechLanguage)
            .putBoolean("contextAwareTranslation",contextAwareTranslation).putBoolean("translationMemoryEnabled",translationMemoryEnabled).putString("glossary",glossary)

            .putInt("subtitlePosition",subtitlePosition.coerceIn(POS_TOP,POS_BOTTOM)).putLong("calibrationMs",calibrationMs.coerceIn(-10000L,10000L)).apply()
    }

    suspend fun load():AppSettings=withContext(Dispatchers.IO){
        val p=prefs ?: return@withContext AppSettings()
        val stored=p.getInt("provider",PROVIDER_MYMEMORY)

import java.io.File

data class WordTimestamp(val word: String, val start: Double, val end: Double)
data class Utterance(val text: String, val start: Double, val end: Double)
data class SttResult(
    val text: String,
    val language: String? = null,
    val words: List<WordTimestamp> = emptyList(),
    val utterances: List<Utterance> = emptyList(),
    val confidence: Double? = null,
    val success: Boolean = true,
    val error: String? = null
)

interface SttClient {
    suspend fun transcribe(file: File, language: String = "auto", contextPrompt: String = ""): SttResult
}

package com.sari.subtitle

import android.util.Log
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.delay
import kotlinx.coroutines.withContext
import okhttp3.HttpUrl.Companion.toHttpUrl
import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.RequestBody.Companion.asRequestBody
import okhttp3.MediaType.Companion.toMediaType
import org.json.JSONObject
import java.io.File
import java.util.concurrent.TimeUnit

class DeepgramClient(
    private val apiKey: String,
    private val http: OkHttpClient = defaultClient()
) : SttClient {
    override suspend fun transcribe(file: File, language: String, contextPrompt: String): SttResult = withContext(Dispatchers.IO) {
        if (apiKey.isBlank()) return@withContext SttResult("", success = false, error = "أدخل مفتاح Deepgram API.")
        if (!file.exists() || file.length() == 0L) return@withContext SttResult("", success = false, error = "الملف الصوتي فارغ.")
        val url = "https://api.deepgram.com/v1/listen".toHttpUrl().newBuilder()
            .addQueryParameter("model", "nova-2")
            .addQueryParameter("language", if (language.isBlank() || language == "auto") "ar" else language)
            .addQueryParameter("smart_format", "true")
            .addQueryParameter("punctuate", "true")
            .addQueryParameter("utterances", "true")
            .addQueryParameter("diarize_model", "latest")
            .apply {
                contextPrompt.split(',', '\n', ';').map { it.trim() }.filter { it.isNotBlank() }.take(20).forEach { addQueryParameter("keywords", it) }
            }
            .build()
        val request = Request.Builder()
            .url(url)
            .header("Authorization", "Token $apiKey")
            .header("Content-Type", "audio/wav")
            .post(file.asRequestBody("audio/wav".toMediaType()))
            .build()
        var last: Exception? = null
        repeat(3) { attempt ->
            try {
                http.newCall(request).execute().use { response ->
                    val raw = response.body?.string().orEmpty()
                    if (!response.isSuccessful) error("Deepgram HTTP ${response.code}: ${raw.take(180)}")
                    return@withContext parse(raw)
                }
            } catch (e: Exception) {
                last = e
                Log.d(TAG, "Deepgram attempt ${attempt + 1} failed: ${e.message}")
                if (attempt < 2) delay(500L * (1L shl attempt))
            }
        }
        SttResult("", success = false, error = "فشل Deepgram بعد 3 محاولات: ${last?.message ?: "unknown"}")
    }

    private fun parse(raw: String): SttResult {
        return try {
            val json = JSONObject(raw)
            val results = json.optJSONObject("results") ?: return SttResult("", success = false, error = "استجابة Deepgram غير صالحة.")
            val channels = results.optJSONArray("channels") ?: return SttResult("", success = false, error = "لم تُرجع Deepgram قنوات صوتية.")
            val alt = channels.optJSONObject(0)?.optJSONArray("alternatives")?.optJSONObject(0)
                ?: return SttResult("", success = false, error = "لم تُرجع Deepgram transcript.")
            val text = alt.optString("transcript").trim()
            val confidence = if (alt.has("confidence")) alt.optDouble("confidence") else null
            val words = mutableListOf<WordTimestamp>()
            alt.optJSONArray("words")?.let { a ->
                for (i in 0 until a.length()) {
                    val w = a.optJSONObject(i) ?: continue
                    val word = w.optString("punctuated_word", w.optString("word")).trim()
                    val start = w.optDouble("start", Double.NaN)
                    val end = w.optDouble("end", Double.NaN)
                    if (word.isNotBlank() && start.isFinite() && end.isFinite()) words += WordTimestamp(word, start, end)
                }
            }
            val utterances = mutableListOf<Utterance>()
            results.optJSONArray("utterances")?.let { a ->
                for (i in 0 until a.length()) {
                    val u = a.optJSONObject(i) ?: continue
                    val t = u.optString("transcript").trim()
                    val start = u.optDouble("start", Double.NaN)
                    val end = u.optDouble("end", Double.NaN)
                    if (t.isNotBlank() && start.isFinite() && end.isFinite()) utterances += Utterance(t, start, end)
                }
            }
            SttResult(text, language = "ar", words = words, utterances = utterances, confidence = confidence)
        } catch (e: Exception) {
            SttResult("", success = false, error = "تعذر تحليل استجابة Deepgram: ${e.message}")
        }
    }

    suspend fun testKey(): Boolean {
        if (apiKey.isBlank()) return false
        return try {
            val request = Request.Builder()
                .url("https://api.deepgram.com/v1/projects")
                .header("Authorization", "Token $apiKey")
                .get().build()
            http.newCall(request).execute().use { it.isSuccessful }
        } catch (e: Exception) {
            Log.d(TAG, "Deepgram key test failed: ${e.message}")
            false
        }
    }

    companion object {
        private const val TAG = "SariDeepgram"
        fun defaultClient() = OkHttpClient.Builder()
            .connectTimeout(20, TimeUnit.SECONDS)
            .readTimeout(30, TimeUnit.SECONDS)
            .callTimeout(35, TimeUnit.SECONDS)
            .build()
    }
}

package com.sari.subtitle

import android.util.Log
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.delay
import kotlinx.coroutines.withContext
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.MultipartBody
import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.RequestBody.Companion.asRequestBody
import org.json.JSONObject
import java.io.File
import java.util.concurrent.TimeUnit

class WhisperClient(private val apiKey: String, private val http: OkHttpClient = defaultClient()) : SttClient {
    override suspend fun transcribe(file: File, language: String, contextPrompt: String): SttResult = withContext(Dispatchers.IO) {
        if (apiKey.isBlank()) return@withContext SttResult("", success = false, error = "أدخل مفتاح Groq API من الإعدادات.")
        if (!file.exists() || file.length() == 0L) return@withContext SttResult("", success = false, error = "الملف الصوتي فارغ.")
        val body = MultipartBody.Builder().setType(MultipartBody.FORM)
            .addFormDataPart("model", "whisper-large-v3-turbo")
            .addFormDataPart("response_format", "verbose_json")
            .addFormDataPart("temperature", "0")
            .apply { if (contextPrompt.isNotBlank()) addFormDataPart("prompt", contextPrompt.take(3_000)) }
            .addFormDataPart("timestamp_granularities[]", "segment")
            .apply { if (language.isNotBlank() && language != "auto") addFormDataPart("language", language) }
            .addFormDataPart("file", file.name, file.asRequestBody("audio/wav".toMediaType())).build()
        val request = Request.Builder().url("https://api.groq.com/openai/v1/audio/transcriptions").header("Authorization", "Bearer $apiKey").post(body).build()
        executeWithRetry(request)
    }

    suspend fun detectLanguage(file: File): String? = withContext(Dispatchers.IO) {
        if (apiKey.isBlank() || !file.exists()) return@withContext null
        val body = MultipartBody.Builder().setType(MultipartBody.FORM)
            .addFormDataPart("model", "whisper-large-v3-turbo")
            .addFormDataPart("response_format", "verbose_json")
            .addFormDataPart("temperature", "0")
            .addFormDataPart("file", file.name, file.asRequestBody("audio/wav".toMediaType())).build()
        val request = Request.Builder().url("https://api.groq.com/openai/v1/audio/transcriptions").header("Authorization", "Bearer $apiKey").post(body).build()
        try {
            http.newCall(request).execute().use { response ->
                if (!response.isSuccessful) return@withContext null
                JSONObject(response.body?.string().orEmpty()).optString("language").ifBlank { null }
            }
        } catch (e: Exception) { Log.d(TAG, "Language detection failed: ${e.message}"); null }
    }

    private suspend fun executeWithRetry(request: Request): SttResult {
        var last: Exception? = null
        repeat(3) { i ->
            try {
                http.newCall(request).execute().use { response ->
                    val raw = response.body?.string().orEmpty()
                    if (!response.isSuccessful) error("Groq HTTP ${response.code}: ${raw.take(180)}")
                    return parse(raw)
                }
            } catch (e: Exception) {
                last = e; Log.d(TAG, "Groq STT attempt ${i + 1} failed: ${e.message}")
                if (i < 2) delay(500L * (i + 1))
            }
        }
        return SttResult("", success = false, error = "فشل Groq STT بعد 3 محاولات: ${last?.message}")
    }

    private fun parse(raw: String): SttResult = try {
        val json = JSONObject(raw)
        val text = json.optString("text").trim()
        val segments = mutableListOf<Utterance>()
        json.optJSONArray("segments")?.let { a -> for (i in 0 until a.length()) { val s=a.optJSONObject(i) ?: continue; val t=s.optString("text").trim(); val st=s.optDouble("start", Double.NaN); val en=s.optDouble("end", Double.NaN); if(t.isNotBlank()&&st.isFinite()&&en.isFinite()) segments += Utterance(t,st,en) } }
        SttResult(text, json.optString("language").ifBlank { null }, utterances = segments, success = true)
    } catch (e: Exception) { SttResult("", success = false, error = "تعذر تحليل Groq: ${e.message}") }

    companion object { private const val TAG="SariWhisper"; fun defaultClient()=OkHttpClient.Builder().connectTimeout(20,TimeUnit.SECONDS).readTimeout(90,TimeUnit.SECONDS).callTimeout(120,TimeUnit.SECONDS).build() }
}

package com.sari.subtitle

import android.util.Log
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import org.json.JSONObject
import org.vosk.Model
import org.vosk.Recognizer
import java.io.File
import java.io.FileInputStream

class VoskClient(private val modelDir: File) : SttClient {
    override suspend fun transcribe(file: File, language: String, contextPrompt: String): SttResult = withContext(Dispatchers.IO) {
        if (!modelDir.exists()) return@withContext SttResult("", success=false, error="نموذج Vosk غير محمّل بعد.")
        try {
            val model = Model(modelDir.absolutePath)
            val recognizer = Recognizer(model, 16_000f)
            FileInputStream(file).use { input ->
                val header = ByteArray(44); input.read(header)
                val buffer = ByteArray(8192)
                while (true) { val n=input.read(buffer); if(n<=0) break; recognizer.acceptWaveForm(buffer,0,n) }
            }
            val finalJson = JSONObject(recognizer.finalResult())
            recognizer.close(); model.close()
            val text=finalJson.optString("text").trim()
            SttResult(text, language.ifBlank { "ar" }, confidence=null, success=text.isNotBlank())
        } catch (e: Exception) { Log.d(TAG,"Vosk failed: ${e.message}"); SttResult("",success=false,error="فشل Vosk: ${e.message}") }
    }
    companion object { private const val TAG="SariVosk" }
}

package com.sari.subtitle

import android.content.Context
import android.util.Log
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.File
import java.net.HttpURLConnection
import java.net.URL
import java.util.zip.ZipInputStream

class ModelDownloader(private val context: Context) {
    suspend fun downloadArabic(onProgress: (Int) -> Unit = {}): File = withContext(Dispatchers.IO) {
        val root=File(context.filesDir,"vosk")
        val model=File(root,"vosk-model-ar-mgb2-0.4")
        if(File(model,"am/final.mdl").exists() || File(model,"conf/mfcc.conf").exists()) return@withContext model
        root.mkdirs(); val zip=File(root,"arabic.zip")
        val url=URL(ARABIC_URL); val conn=url.openConnection() as HttpURLConnection
        conn.connectTimeout=20_000; conn.readTimeout=60_000
        try {
            val total=conn.contentLengthLong
            conn.inputStream.use { input -> zip.outputStream().use { out -> val b=ByteArray(32*1024); var read=0L; while(true){val n=input.read(b); if(n<0)break; out.write(b,0,n); read+=n; if(total>0) onProgress(((read*100)/total).toInt().coerceIn(0,100))} } }
            ZipInputStream(zip.inputStream().buffered()).use { zin ->
                var entry=zin.nextEntry
                while(entry!=null){ val safe=entry.name.replace("\\","/"); if(safe.contains("..")) {entry=zin.nextEntry; continue}; val out=File(root,safe); if(entry.isDirectory) out.mkdirs() else {out.parentFile?.mkdirs(); out.outputStream().use{zin.copyTo(it)}}; zin.closeEntry(); entry=zin.nextEntry }
            }
            val extracted=File(root,"vosk-model-ar-mgb2-0.4"); if(!extracted.exists()) error("لم يتم استخراج نموذج Vosk بشكل صحيح.")
            zip.delete(); onProgress(100); extracted
        } catch(e:Exception){ Log.d(TAG,"Model download failed: ${e.message}"); zip.delete(); throw e } finally { conn.disconnect() }
    }
    fun modelDirectory(): File=File(context.filesDir,"vosk/vosk-model-ar-mgb2-0.4")
    companion object { const val ARABIC_URL="https://alphacephei.com/vosk/models/vosk-model-ar-mgb2-0.4.zip"; private const val TAG="SariModel" }
}

class OpenAiClient(
    private val apiKey: String,
    private val http: OkHttpClient = defaultClient()
) {
    suspend fun translate(text: String, source: String, target: String, context: TranslationContext): String = withContext(Dispatchers.IO) {
        require(apiKey.isNotBlank()) { "أدخل مفتاح OpenAI API في الإعدادات." }
        val system = buildString {
            append("أنت مترجم محترف. ترجم النص إلى العربية بأسلوب طبيعي وواضح. حافظ على الأسماء والمصطلحات. لا تُضف معلومات غير موجودة. اجعل الترجمة مختصرة ومناسبة للترجمة المرئية. ")
            append("لغة المصدر: ").append(if (source.isBlank()) "auto" else source).append(". الهدف: ").append(target).append(". ")
            append("لا تشرح ولا تضع ملاحظات؛ أعد الترجمة فقط.\n")
            append(contextText(context))
        }
        val payload = JSONObject()
            .put("model", "gpt-4o-mini")
            .put("temperature", 0.3)
            .put("max_tokens", 500)
            .put("messages", JSONArray()
                .put(JSONObject().put("role", "system").put("content", system))
                .put(JSONObject().put("role", "user").put("content", text)))
            .toString()
        val request = Request.Builder().url("https://api.openai.com/v1/chat/completions")
            .header("Authorization", "Bearer $apiKey")
            .header("Content-Type", "application/json")
            .post(payload.toRequestBody("application/json; charset=utf-8".toMediaType()))
            .build()
        var last: Exception? = null
        repeat(3) { i ->
            try {
                return@withContext http.newCall(request).execute().use { response ->
                    val raw = response.body?.string().orEmpty()
                    if (!response.isSuccessful) error("OpenAI HTTP ${response.code}: ${raw.take(180)}")
                    val result = JSONObject(raw).getJSONArray("choices").getJSONObject(0).getJSONObject("message").getString("content").trim()
                    if (result.isBlank()) error("OpenAI أعاد ترجمة فارغة.")
                    result
                }
            } catch (e: Exception) {
                last = e
                Log.d(TAG, "OpenAI attempt ${i + 1} failed: ${e.message}")
                if (i < 2) delay(500L * (i + 1))
            }
        }
        throw IllegalStateException("فشلت ترجمة OpenAI بعد 3 محاولات.", last)
    }

    suspend fun testKey(): Boolean = try {
        if (apiKey.isBlank()) false else http.newCall(Request.Builder().url("https://api.openai.com/v1/models/gpt-4o-mini").header("Authorization", "Bearer $apiKey").get().build()).execute().use { it.isSuccessful }
    } catch (e: Exception) { Log.d(TAG, "OpenAI key test failed: ${e.message}"); false }

    private fun contextText(context: TranslationContext): String = buildString {
        if (context.previousTranslations.isNotEmpty()) {
            append("السياق السابق للاختيار اللغوي فقط:\n")
            context.previousTranslations.takeLast(6).forEach { append("SRC: ${it.source}\nTGT: ${it.translation}\n") }
        }
        if (context.glossary.isNotEmpty()) append("القاموس الإلزامي:\n${context.glossary.entries.joinToString("\n") { "${it.key} = ${it.value}" }}\n")
        if (context.characterNotes.isNotBlank()) append("ملاحظات الأسماء/المتحدثين:\n${context.characterNotes}\n")
    }

    companion object {
        private const val TAG = "SariOpenAI"
        fun defaultClient() = OkHttpClient.Builder().connectTimeout(20, TimeUnit.SECONDS).readTimeout(30, TimeUnit.SECONDS).callTimeout(35, TimeUnit.SECONDS).build()
    }
}

package com.sari.subtitle

import android.content.Context
import android.content.Intent
import android.os.SystemClock
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.File

class TranslationRepository(private val context:Context){
    private val deduplicator=TranscriptDeduplicator()
    private val memory by lazy{AppDatabase.get(context).translationMemoryDao()}
    private val episodeContext by lazy{EpisodeContextStore(context)}

    suspend fun translateAudio(audioFile:File,capturedAtElapsedMs:Long=SystemClock.elapsedRealtime(),durationMs:Long=AudioChunker.CHUNK_MS):TranslationEntity=withContext(Dispatchers.IO){
        val settings=SettingsRepository(context).load()
        val pre=if(settings.contextAwareTranslation)episodeContext.snapshot() else TranslationContext()
        val prompt=buildPrompt(settings,pre)
        val stt=createStt(settings)
        val speechResult=stt.transcribe(audioFile,settings.speechLanguage,prompt)
        if(!speechResult.success) throw IllegalStateException(speechResult.error ?: "فشل تحويل الصوت إلى نص.")
        val speech=speechResult.text.trim()
        val sourceLanguage=if(settings.sourceLanguage=="auto") speechResult.language ?: "auto" else settings.sourceLanguage
        val target=settings.targetLanguage.ifBlank{"ar"}
        if(speech.isBlank()||!deduplicator.shouldTranslate(speech)) return@withContext TranslationEntity(sourceText=speech,translatedText="",sourceLanguage=sourceLanguage,targetLanguage=target,createdAt=System.currentTimeMillis())
        val hash=TextHash.sha256(speech)
        val cached=if(settings.translationMemoryEnabled)memory.find(hash,sourceLanguage,target) else null
        val context=if(settings.contextAwareTranslation)episodeContext.snapshot() else TranslationContext()
        val translated=cached?.translatedText ?: run{
            val quality=when(settings.translationQuality){0->TranslationQuality.FAST;1->TranslationQuality.BALANCED;else->TranslationQuality.HUMAN}
            val provider=when(settings.provider){SettingsRepository.PROVIDER_GROQ->TranslationProvider.GROQ;SettingsRepository.PROVIDER_GOOGLE->TranslationProvider.GOOGLE;SettingsRepository.PROVIDER_OPENAI->TranslationProvider.OPENAI;else->TranslationProvider.MYMEMORY}
            val key=when(provider){TranslationProvider.GROQ->settings.speechKey;TranslationProvider.OPENAI->settings.openAiKey;else->settings.translationKey}
            val result=TranslationClient(provider,key,settings.myMemoryEmail).translateWithContext(speech,sourceLanguage,target,context,quality)
            if(settings.translationMemoryEnabled){memory.insert(TranslationMemoryEntity(sourceHash=hash,sourceText=speech,translatedText=result,sourceLanguage=sourceLanguage,targetLanguage=target,createdAt=System.currentTimeMillis()));memory.trim()}
            result
        }
        episodeContext.addTurn(speech,translated)
        val startOffset=(speechResult.utterances.minOfOrNull{it.start}?.times(1000)?.toLong() ?: speechResult.words.minOfOrNull{it.start}?.times(1000)?.toLong() ?: 0L)
        val endOffset=(speechResult.utterances.maxOfOrNull{it.end}?.times(1000)?.toLong() ?: speechResult.words.maxOfOrNull{it.end}?.times(1000)?.toLong() ?: durationMs.toDouble()).toLong()
        val start=capturedAtElapsedMs+startOffset+settings.calibrationMs
        val end=capturedAtElapsedMs+endOffset+settings.calibrationMs
        val cueDuration=(end-start).coerceIn(900L,6000L)
        LiveSubtitleBus.publish(speech,translated,start,cueDuration)
        runCatching{context.startService(Intent(context,OverlayService::class.java).setAction(OverlayService.ACTION_UPDATE).putExtra(OverlayService.EXTRA_ORIGINAL,speech).putExtra(OverlayService.EXTRA_TRANSLATED,translated).putExtra(OverlayService.EXTRA_DURATION_MS,cueDuration))}
        val entity=TranslationEntity(sourceText=speech,translatedText=translated,sourceLanguage=sourceLanguage,targetLanguage=target,createdAt=System.currentTimeMillis())
        AppDatabase.get(context).translationDao().insert(entity)
        entity
    }

    private fun createStt(settings:AppSettings):SttClient=when(settings.sttProvider){
        SettingsRepository.STT_DEEPGRAM->DeepgramClient(settings.deepgramKey)
        SettingsRepository.STT_VOSK->VoskClient(ModelDownloader(context).modelDirectory())
        else->WhisperClient(settings.speechKey)
    }
    private fun buildPrompt(settings:AppSettings,c:TranslationContext)=buildString{if(settings.contextPrompt.isNotBlank())append(settings.contextPrompt).append("\n");if(c.glossary.isNotEmpty())append("Known terminology: ").append(c.glossary.entries.joinToString(", "){"${it.key}=${it.value}"}).append(". ");if(c.previousSegments.isNotEmpty())append("Recent dialogue for spelling/context: ").append(c.previousSegments.takeLast(4).joinToString(" | "))}.take(3000)
    fun resetEpisodeContext()=episodeContext.clearSession()
    suspend fun clearTranslationMemory()=withContext(Dispatchers.IO){memory.clear()}
}

package com.sari.subtitle

import android.content.ContentValues
import android.content.Context
import android.os.Environment
import android.provider.MediaStore
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

object SrtExporter {
    suspend fun export(context: Context, cues: List<SubtitleCue>): String? = withContext(Dispatchers.IO) {
        if (cues.isEmpty()) return@withContext null
        val name="SariSubtitle_"+SimpleDateFormat("yyyyMMdd_HHmmss",Locale.US).format(Date())+".srt"
        val content=cues.mapIndexed { i,c -> "${i+1}\n${format(c.startElapsedMs)} --> ${format(c.endElapsedMs)}\n${c.translated.trim()}\n" }.joinToString("\n")
        if (android.os.Build.VERSION.SDK_INT >= 29) {
            val values=ContentValues().apply { put(MediaStore.Downloads.DISPLAY_NAME,name); put(MediaStore.Downloads.MIME_TYPE,"application/x-subrip"); put(MediaStore.Downloads.RELATIVE_PATH,Environment.DIRECTORY_DOWNLOADS+"/SariSubtitle/") }
            val uri=context.contentResolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI,values) ?: return@withContext null
            context.contentResolver.openOutputStream(uri)?.use{it.write(content.toByteArray(Charsets.UTF_8))} ?: return@withContext null
            uri.toString()
        } else null
    }
    private fun format(ms:Long):String { val x=ms.coerceAtLeast(0); return String.format(Locale.US,"%02d:%02d:%02d,%03d",x/3600000,(x/60000)%60,(x/1000)%60,x%1000) }
}
